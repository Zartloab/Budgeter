function getWeekStart(d=new Date()) {
  const x=new Date(d), day=x.getDay(), diff=x.getDate()-day+(day===0?-6:1);
  x.setDate(diff); x.setHours(0,0,0,0); return x;
}
function getMonthStart(d=new Date()) { return new Date(d.getFullYear(),d.getMonth(),1); }
function parseDate(s) { const [y,m,d]=String(s).split('-').map(Number); return new Date(y,m-1,d); }
function txInPeriod(tx, p) {
  if (p==='all') return true;
  const d=parseDate(tx.date), now=new Date();
  if (p==='week') return d>=getWeekStart();
  if (p==='month') return d>=getMonthStart();
  if (p==='last30') { const c=new Date(); c.setDate(c.getDate()-30); return d>=c; }
  return true;
}

// ═══════════════════════ COMPUTED ═══════════════════════
function C() {
  const billsTotal = cfg.billItems.length>0 ? cfg.billItems.reduce((s,i)=>s+i.amount,0) : cfg.weeklyBills;
  const weeklyAvg = cfg.myWeeklyIncome + cfg.wifeFortnightlyIncome/2;
  const eDone = data.emergencyBalance >= cfg.emergencyTarget;
  const dDone = data.debtPaid >= cfg.totalDebt;
  const phase = dDone?3:eDone?2:1;
  const phaseC = ['','#54A0FF','#FF9F43','#55EFC4'][phase];
  const eAlloc = eDone?0:cfg.weeklyEmergency;
  const fixed = cfg.weeklyRent+billsTotal+eAlloc+cfg.weeklyGoals;
  const debtAlloc = !dDone?Math.max(0,weeklyAvg-fixed-cfg.weeklySpends):0;
  const savAlloc = dDone?Math.max(0,weeklyAvg-fixed-cfg.weeklySpends):0;
  const remDebt = Math.max(0,cfg.totalDebt-data.debtPaid);
  const weeksLeft = debtAlloc>0?Math.ceil(remDebt/debtAlloc):null;
  const monthlyIncome = weeklyAvg*4.33;

  const expenses = data.transactions.filter(t=>t.type!=='income'&&t.type!=='transfer');
  const ws=getWeekStart(), ms=getMonthStart();
  const weekExp = expenses.filter(t=>parseDate(t.date)>=ws);
  const monthExp = expenses.filter(t=>parseDate(t.date)>=ms);
  const weekTotal = weekExp.reduce((s,t)=>s+t.amount,0);
  const monthTotal = monthExp.reduce((s,t)=>s+t.amount,0);

  const monthByCat = {};
  CATS.forEach(c=>monthByCat[c]=0);
  monthExp.forEach(t=>{ if(monthByCat[t.category]!==undefined) monthByCat[t.category]+=t.amount; else monthByCat['Other']+=t.amount; });

  const savingsRate = monthlyIncome>0?Math.max(0,Math.round((1-monthTotal/monthlyIncome)*100)):0;
  return {billsTotal,weeklyAvg,eDone,dDone,phase,phaseC,eAlloc,fixed,debtAlloc,savAlloc,remDebt,weeksLeft,monthlyIncome,weekTotal,monthTotal,monthByCat,savingsRate};
}

// ═══════════════════════ TRANSACTIONS ═══════════════════════
function addTx(tx) { data.transactions.unshift({id:uid(),...tx}); saveData(); }
function updateTx(id, upd) { const i=data.transactions.findIndex(t=>t.id===id); if(i>=0){data.transactions[i]={...data.transactions[i],...upd};saveData();} }
function deleteTx(id) { data.transactions=data.transactions.filter(t=>t.id!==id); saveData(); }

function getFilteredTx() {
  return data.transactions
    .filter(t => {
      if (txFilter.search && !`${t.merchant}${t.category}${t.notes||''}`.toLowerCase().includes(txFilter.search.toLowerCase())) return false;
      if (txFilter.category && t.category!==txFilter.category) return false;
      if (!txInPeriod(t, txFilter.period)) return false;
      return true;
    })
    .sort((a,b) => {
      if (txSort.field==='date') return (a.date<b.date?-1:1)*txSort.dir;
      if (txSort.field==='amount') return (a.amount-b.amount)*txSort.dir;
      return 0;
    });
}

  const merchant=($('tx-merchant')?.value||'').trim();
  const amount=parseFloat($('tx-amount')?.value);
  const cat=$('tx-cat')?.value||'Other';
  const date=$('tx-date')?.value||today();
  const notes=($('tx-notes')?.value||'').trim();

function renderAnalytics() {
  const {monthByCat,weekTotal,monthTotal,weeklyAvg}=C();
  const weeks=[];
  for(let i=7;i>=0;i--){
    const ws=getWeekStart(new Date(Date.now()-i*7*864e5));
    const we=new Date(ws); we.setDate(we.getDate()+7);
    const tot=data.transactions.filter(t=>t.type!=='income'&&t.type!=='transfer').filter(t=>{const d=parseDate(t.date);return d>=ws&&d<we;}).reduce((s,t)=>s+t.amount,0);
    weeks.push({label:`${ws.getDate()}/${ws.getMonth()+1}`,total:tot});

async function callClaude(system, user, maxTok=1200) {
  if (!cfg.claudeApiKey) { flash('Add Claude API key in Settings', false); return null; }
  try {
    const r = await fetch('https://api.anthropic.com/v1/messages', {
      method:'POST',
      headers:{'Content-Type':'application/json','x-api-key':cfg.claudeApiKey,'anthropic-version':'2023-06-01'},
      body:JSON.stringify({model:CLAUDE_MODEL,max_tokens:maxTok,system,messages:[{role:'user',content:user}]})
    });
    if (!r.ok) { const e=await r.json(); throw new Error(e.error?.message||`HTTP ${r.status}`); }
    const d=await r.json();
    return d.content?.[0]?.text||'';
  } catch(e) { flash('AI error: '+e.message, false); return null; }
}

window.processBank = async () => {
  const text=($('bank-text')?.value||'').trim();
  const date=$('bank-date')?.value||today();
  if(!text){flash('Paste some transaction text',false);return;}
  const btn=document.querySelector('#modal .btn');
  if(btn){btn.disabled=true;btn.textContent='⏳ Processing...';}
  const reply=await callClaude(`Parse bank statement text into transactions. Return ONLY a valid JSON array. Each item: {"merchant":"string","amount":number,"category":"one of: ${CATS.join(',')}"}. Amount always positive. Infer category from merchant. No markdown, no explanation, only the JSON array.`,text,1500);
  if(!reply){if(btn){btn.disabled=false;btn.textContent='🤖 Categorise with AI →';}return;}
  try {
    const clean=reply.replace(/```json?|```/g,'').trim();
    const txs=JSON.parse(clean);
    if(!Array.isArray(txs)) throw new Error('Not array');
    let count=0;
    txs.forEach(t=>{if(t.merchant&&t.amount>0){addTx({merchant:t.merchant,amount:parseFloat(t.amount),category:CATS.includes(t.category)?t.category:'Other',date,notes:'AI import',recurring:false,type:'expense'});count++;}});
    closeModal(); flash(`✓ ${count} transactions imported!`); renderAll();
  } catch(e) { flash('Could not parse AI response. Try again.',false); if(btn){btn.disabled=false;btn.textContent='🤖 Categorise with AI →';} }
};

// ═══════════════════════ EXPORT / IMPORT ═══════════════════════
function download(content, name, type) {
  const b=new Blob([content],{type}), u=URL.createObjectURL(b), a=document.createElement('a');
  a.href=u; a.download=name; a.click(); URL.revokeObjectURL(u);
}

window.exportJSON = () => {
  download(JSON.stringify({version:2,exportedAt:new Date().toISOString(),cfg,data},null,2), `finos-backup-${today()}.json`, 'application/json');
  flash('Backup exported ✓');
};

window.importJSON = inp => {
  const f=inp.files[0]; if(!f)return;
  const r=new FileReader();
  r.onload=e=>{
    try {
      const b=JSON.parse(e.target.result);
      if(!b.cfg||!b.data) throw new Error('Invalid file');
      if(!confirm(`Import backup from ${b.exportedAt}?\nThis will overwrite all current data.`))return;
      cfg={...CFG0,...b.cfg}; data={...DATA0,...b.data};
      saveCfg(); saveData(); renderAll(); flash('Backup imported ✓');
    } catch(err){flash('Import failed: '+err.message,false);}
  };
  r.readAsText(f); inp.value='';
};

window.exportTxCSV = () => {
  if(!data.transactions.length){flash('No transactions to export',false);return;}
  const h='Date,Merchant,Amount,Category,Type,Recurring,Notes';
  const rows=data.transactions.map(t=>`${t.date},"${(t.merchant||'').replace(/"/g,'""')}",${t.amount},"${t.category}","${t.type||'expense'}",${t.recurring?'Yes':'No'},"${(t.notes||'').replace(/"/g,'""')}"`);
  download([h,...rows].join('\n'), `finos-transactions-${today()}.csv`, 'text/csv');
  flash('CSV exported ✓');
};

window.exportSummaryCSV = () => {
  const {monthByCat}=C();
  const h='Category,Weekly Budget,Monthly Budget,Spent This Month,Remaining,Status';
  const rows=Object.entries(cfg.categoryBudgets).map(([cat,wk])=>{
    const mo=(wk*4.33).toFixed(2), sp=(monthByCat[cat]||0).toFixed(2), rem=(wk*4.33-monthByCat[cat]||0).toFixed(2);
    const st=wk===0?'No budget':parseFloat(sp)>parseFloat(mo)?'Over':'On track';
    return `"${cat}",${wk},${mo},${sp},${rem},"${st}"`;
  });
  download([h,...rows].join('\n'), `finos-summary-${today()}.csv`, 'text/csv');
  flash('Summary exported ✓');
};
